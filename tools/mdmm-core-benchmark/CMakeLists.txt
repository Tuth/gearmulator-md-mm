cmake_minimum_required(VERSION 3.16)
project(mdmmCoreBenchmark LANGUAGES C CXX)
get_filename_component(GEARMULATOR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${GEARMULATOR_ROOT}/base.cmake")
include(CTest)
option(gearmulator_DIAGNOSTICS "Compile opt-in diagnostic timings" OFF)
if(gearmulator_DIAGNOSTICS)
  add_compile_definitions(GEARMULATOR_DIAGNOSTIC_LOGGING=1)
else()
  add_compile_definitions(GEARMULATOR_DIAGNOSTIC_LOGGING=0)
endif()
set(ASMJIT_STATIC TRUE)
set(ASMJIT_NO_INSTALL TRUE)
set(BUILD_SHARED_LIBS OFF)
foreach(component dsp56300/source baseLib libresample synthLib mc68k hardwareLib elektron/md/mdLib)
  string(REPLACE "/" "_" binary_dir "${component}")
  add_subdirectory("${GEARMULATOR_ROOT}/source/${component}" "${binary_dir}" EXCLUDE_FROM_ALL)
endforeach()
add_executable(mdmmCoreBenchmark benchmark.cpp)
target_link_libraries(mdmmCoreBenchmark PRIVATE mdLib)
set(firmware_tests mdAudioFirmwareTest mmAudioFirmwareTest idleSchedulerFirmwareTest)
if(MD_TRANSPORT_DIAGNOSTICS)
  list(APPEND firmware_tests transportScorecardFirmwareTest)
endif()
foreach(test IN LISTS firmware_tests)
  add_executable(${test} "${GEARMULATOR_ROOT}/source/elektron/md/mdLibTest/${test}.cpp")
  target_link_libraries(${test} PRIVATE mdLib)
  add_test(NAME ${test} COMMAND ${test})
  set_tests_properties(${test} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 300)
endforeach()

foreach(target mdmmCoreBenchmark mdAudioFirmwareTest mmAudioFirmwareTest)
  if(MSVC)
    target_compile_options(${target} PRIVATE /fp:precise)
  else()
    target_compile_options(${target} PRIVATE -fno-fast-math)
  endif()
endforeach()
